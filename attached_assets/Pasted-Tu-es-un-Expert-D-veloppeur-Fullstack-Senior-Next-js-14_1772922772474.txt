A. Transition Fin de Rencontre (Le Cœur du système)
Cette fonction gère le passage à DONE, la transition de la Relation vers ACTIVE (si c'est la 1ère), et l'activation du mode Feedback.

typescript

import { prisma } from '@/lib/prisma';

export async function markInterventionDone(interventionId: string) {
  return await prisma.$transaction(async (tx) => {
    // 1. Mettre à jour l'intervention
    const intervention = await tx.intervention.update({
      where: { id: interventionId },
      data: { status: 'DONE' },
      include: { relation: true }
    });
    
    // 2. Logique de "Cadrage" : Si la relation était en attente de 1er RDV
    if (intervention.relation.status === 'PENDING_FIRST_MEETING') {
      await tx.relation.update({
        where: { id: intervention.relationId },
        data: { status: 'ACTIVE' }
      });
      
      // Notifier les deux que c'est débloqué
      // (Code de notification à ajouter ici)
    }
    
    // 3. Passer en attente de feedback (Gatekeeping)
    const updated = await tx.intervention.update({
      where: { id: interventionId },
      data: { status: 'COMPLETED_PENDING_FEEDBACK' }
    });

    return updated;
  });
}
B. Soumission de Feedback
Cette fonction vérifie que les deux ont noté avant de clôturer.

typescript

export async function submitFeedback(
  interventionId: string, 
  userId: string, 
  rating: number, 
  comment?: string
) {
  const intervention = await prisma.intervention.findUnique({
    where: { id: interventionId },
    include: { relation: true }
  });
  
  if (!intervention || intervention.status !== 'COMPLETED_PENDING_FEEDBACK') {
    throw new Error('Feedback non autorisé ou déjà complété');
  }
  
  const isSenior = intervention.relation.seniorId === userId;
  
  // Mise à jour du feedback concerné
  const updated = await prisma.intervention.update({
    where: { id: interventionId },
    data: isSenior 
      ? { seniorRating: rating, seniorComment: comment }
      : { juniorRating: rating, juniorComment: comment }
  });
  
  // Vérification : Si les deux ont noté -> Clôturer
  if (updated.seniorRating !== null && updated.juniorRating !== null) {
    await prisma.intervention.update({
      where: { id: interventionId },
      data: { status: 'COMPLETED' }
    });
  }
  
  return updated;
}

export async function checkAndSendFeedbackReminders(userId: string) {
  // Chercher les interventions DONE depuis plus de 3 jours sans rappel envoyé
  const threeDaysAgo = new Date();
  threeDaysAgo.setDate(threeDaysAgo.getDate() - 3);

  const pending = await prisma.intervention.findMany({
    where: {
      status: 'COMPLETED_PENDING_FEEDBACK',
      feedbackReminderSentAt: null,
      createdAt: { lte: threeDaysAgo },
      relation: {
        OR: [
          { seniorId: userId },
          { juniorId: userId }
        ]
      }
    }
  });

  // Pour chaque intervention trouvée, envoyer email Brevo et marquer le rappel envoyé
  pending.forEach(async (intervention) => {
    // TODO: Appeler fonction Brevo sendEmail(...)
    await prisma.intervention.update({
      where: { id: intervention.id },
      data: { feedbackReminderSentAt: new Date() }
    });
  });
}
